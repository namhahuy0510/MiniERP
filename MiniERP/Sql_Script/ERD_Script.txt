Table AspNetUsers {
  UserId int [pk] // Khóa chính
  TenDangNhap varchar
  Email varchar
  MatKhauHash varchar
}

Table AspNetRoles {
  RoleId int [pk]
  TenVaiTro varchar
}

Table AspNetUserRoles {
  UserId int [ref: > AspNetUsers.UserId]
  RoleId int [ref: > AspNetRoles.RoleId]
}

Table AspNetUserClaims {
  Id int [pk]
  UserId int [ref: > AspNetUsers.UserId]
  LoaiClaim varchar
  GiaTriClaim varchar
}

Table AspNetRoleClaims {
  Id int [pk]
  RoleId int [ref: > AspNetRoles.RoleId]
  LoaiClaim varchar
  GiaTriClaim varchar
}

Table AspNetUserLogins {
  LoginProvider varchar
  ProviderKey varchar
  UserId int [ref: > AspNetUsers.UserId]
}

Table AspNetUserTokens {
  UserId int [ref: > AspNetUsers.UserId]
  LoginProvider varchar
  TenToken varchar
  GiaTri varchar
}

Table Departments {
  DepartmentId int [pk]
  TenPhongBan varchar
}

Table Employees {
  EmployeeId int [pk]
  HoTen varchar
  DepartmentId int [ref: > Departments.DepartmentId]
}

Table Attendance {
  AttendanceId int [pk]
  EmployeeId int [ref: > Employees.EmployeeId]
  Ngay date
  TrangThai varchar
}

Table __EFMigrationsHistory {
  MigrationId varchar [pk]
  PhienBanSanPham varchar
}
